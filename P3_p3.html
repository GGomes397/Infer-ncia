Prova 3-par

Avaliação do desempenho de algumas estatísticas de Teste

No trabalho serão avaliados três estatísticas a de Wald, Escore e a Razão de Verossimilhanças. Os métodos que usaremos para avaliar essas estatísticas serão em relação ao tamanho das amostras escolhidos, poder do teste e iremos estimar o nível de significância pelo Método de Monte Carlo. O nível de significância nominal é de 5%.

Tamanho do teste

set.seed(123) 
n_grid <- c(10, 50, 100, 500, 1000, 5000, 10000)
B <- 5000
alpha <- 0.05
h0 <- 5
lambda <- 1
vcrit <- qchisq(1 - alpha, df = 1)

# CORREÇÃO 1: Estimador de máxima verossimilhança usando optimize()
estimar_beta_weibull <- function(amostra) {
  n <- length(amostra)
  log_amostra <- log(amostra)
  sum_log <- sum(log_amostra)
  
  # Função de log-verossimilhança negativa
  log_veross <- function(beta) {
    if (beta <= 0) return(1e10)
    -(n * log(beta) + (beta - 1) * sum_log - sum(amostra^beta))
  }
  
  # Otimização numérica
  resultado <- tryCatch(
    optimize(log_veross, interval = c(0.1, 50), maximum = FALSE),
    error = function(e) NULL
  )
  
  if (!is.null(resultado)) {
    return(resultado$minimum)
  } else {
    return(NA)
  }
}

calcular_estatisticas <- function(amostra, beta0) {
  n_obs <- length(amostra)
  
  # CORREÇÃO 2: Usar a nova função de estimação
  beta_hat <- tryCatch(estimar_beta_weibull(amostra), error = function(e) NA)
  
  if (is.na(beta_hat) || beta_hat <= 0 || !is.finite(beta_hat)) {
    return(c(qw = NA, qs = NA, qr = NA))
  }
  
  log_amostra <- log(amostra)
  sum_log <- sum(log_amostra)
  
  # 1. Wald
  qw <- n_obs * (beta_hat - beta0)^2 / (beta0^2)
  
  # CORREÇÃO 3: Função escore correta
  # U(beta0) = n/beta0 + sum(log(x)) - sum(x^beta0 * log(x))
  x_beta0 <- amostra^beta0
  U_beta0 <- n_obs / beta0 + sum_log - sum(x_beta0 * log_amostra)
  
  # Informação de Fisher em beta0
  I_beta0 <- n_obs / (beta0^2)
  qs <- (U_beta0^2) / I_beta0
  
  # 3. Razão de Verossimilhanças (RV)
  logL_hat <- n_obs * log(beta_hat) + (beta_hat - 1) * sum_log - sum(amostra^beta_hat)
  logL_h0  <- n_obs * log(beta0) + (beta0 - 1) * sum_log - sum(amostra^beta0)
  qr <- 2 * (logL_hat - logL_h0)
  
  # CORREÇÃO 4: Garantir que qr não seja negativo devido a problemas numéricos
  if (qr < 0) qr <- 0
  
  return(c(qw = qw, qs = qs, qr = qr))
}

tamanho_resultados <- matrix(0, nrow = length(n_grid), ncol = 3)
colnames(tamanho_resultados) <- c("Wald", "Escore", "RV")

for (i in seq_along(n_grid)) {
  n_val <- n_grid[i]
  rejeicoes <- c(Wald = 0, Escore = 0, RV = 0)
  
  for (j in 1:B) {
    amostra <- rweibull(n_val, shape = h0, scale = lambda)
    stats <- calcular_estatisticas(amostra, h0)
    
    if (!any(is.na(stats))) {
      if (stats["qw"] > vcrit) rejeicoes["Wald"]   <- rejeicoes["Wald"] + 1
      if (stats["qs"] > vcrit) rejeicoes["Escore"] <- rejeicoes["Escore"] + 1
      if (stats["qr"] > vcrit) rejeicoes["RV"]     <- rejeicoes["RV"] + 1
    }
  }
  tamanho_resultados[i, ] <- rejeicoes / B
}

df_tamanho <- data.frame(n = n_grid, tamanho_resultados)

Poder do teste

beta_alternativos <- c(5.5, 6.0, 7.0, 8.0, 10.0)

poder_Wald   <- matrix(0, nrow = length(n_grid), ncol = length(beta_alternativos))
poder_Escore <- matrix(0, nrow = length(n_grid), ncol = length(beta_alternativos))
poder_RV     <- matrix(0, nrow = length(n_grid), ncol = length(beta_alternativos))

colnames(poder_Wald)   <- paste0("beta=", beta_alternativos)
colnames(poder_Escore) <- paste0("beta=", beta_alternativos)
colnames(poder_RV)     <- paste0("beta=", beta_alternativos)

for (k in seq_along(beta_alternativos)) {
  beta_alt <- beta_alternativos[k]
  
  for (i in seq_along(n_grid)) {
    n_val <- n_grid[i]
    rej <- c(Wald = 0, Escore = 0, RV = 0)
    
    for (j in 1:B) {
      amostra <- rweibull(n_val, shape = beta_alt, scale = lambda)
      stats <- calcular_estatisticas(amostra, h0) # Testando contra H0: beta = 5
      
      if (!any(is.na(stats))) {
        if (stats["qw"] > vcrit) rej["Wald"]   <- rej["Wald"] + 1
        if (stats["qs"] > vcrit) rej["Escore"] <- rej["Escore"] + 1
        if (stats["qr"] > vcrit) rej["RV"]     <- rej["RV"] + 1
      }
    }
    
    poder_Wald[i, k]   <- rej["Wald"] / B
    poder_Escore[i, k] <- rej["Escore"] / B
    poder_RV[i, k]     <- rej["RV"] / B
  }
}

n_grande <- 10000
B_ks <- 5000

estatisticas_assint <- matrix(NA,
                              nrow = B_ks,
                              ncol = 3)

colnames(estatisticas_assint) <- c("Wald","Escore","RV")

for(j in 1:B_ks){

  amostra <- rweibull(n_grande,
                      shape = h0,
                      scale = lambda)

  res <- calcular_estatisticas(amostra,h0)

  if(all(is.finite(res))){
    estatisticas_assint[j,] <- res
  }

}

estatisticas_assint <-
  estatisticas_assint[
    complete.cases(estatisticas_assint),
  ]

    nrow(estatisticas_assint),"\n")
